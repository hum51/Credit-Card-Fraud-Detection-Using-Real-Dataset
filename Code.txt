                             f1_score, roc_auc_score, confusion_matrix,
                             classification_report, roc_curve)
from imblearn.over_sampling import SMOTE
import warnings

print("="*60)
print("CREDIT CARD FRAUD DETECTION SYSTEM")

print("\n Downloading actual Credit Card Fraud Dataset...")

print(" Dataset downloaded successfully!")
print("\n Loading dataset...")
df = pd.read_csv('creditcard.csv')

print(f"\n Dataset loaded successfully!")
print(f"   Total transactions: {len(df):,}")
print(f"   Features: {df.shape[1]}")
print(f"   Memory usage: {df.memory_usage().sum() / 1024**2:.2f} MB")

print(f"\n Class Distribution:")
print(f"   Legitimate (Class 0): {(df['Class']==0).sum():,} ({(df['Class']==0).mean()*100:.4f}%)")
print(f"   Fraudulent (Class 1): {(df['Class']==1).sum():,} ({(df['Class']==1).mean()*100:.4f}%)")
print(f"   Imbalance Ratio: {((df['Class']==0).sum()/(df['Class']==1).sum()):.1f}:1")


print("\n Preprocessing data...")
print(f"Missing values: {df.isnull().sum().sum()}")

scaler = RobustScaler()
df['Scaled_Amount'] = scaler.fit_transform(df[['Amount']])
df['Scaled_Time'] = scaler.fit_transform(df[['Time']])

df['Hour'] = (df['Time'] / 3600) % 24

X = df.drop(['Class', 'Time', 'Amount'], axis=1)
y = df['Class']

print(f"Feature shape: {X.shape}")
print(f"Target shape: {y.shape}")

print("\n Splitting data...")
X_train, X_test, y_train, y_test = train_test_split(
    X, y, test_size=0.3, random_state=42, stratify=y
)

print(f"Training set: {len(X_train):,} transactions")
print(f"Test set: {len(X_test):,} transactions")
print(f"Training fraud ratio: {y_train.mean()*100:.4f}%")
print(f"Test fraud ratio: {y_test.mean()*100:.4f}%")


print("\n⚖️ Handling class imbalance with SMOTE...")
print(f"Before SMOTE - Class distribution: {y_train.value_counts().to_dict()}")

smote = SMOTE(random_state=42, sampling_strategy='auto')
X_train_resampled, y_train_resampled = smote.fit_resample(X_train, y_train)

print(f"After SMOTE - Class distribution: {pd.Series(y_train_resampled).value_counts().to_dict()}")

print("\n Training Random Forest model...")
print("(Ismein 2-3 minute lag sakte hain, please wait...)")

model = RandomForestClassifier(
    n_estimators=100,
    max_depth=15,
    min_samples_split=5,
    random_state=42,
    n_jobs=-1,
    verbose=0
)

model.fit(X_train_resampled, y_train_resampled)
print(" Model training complete!")

print("\n Making predictions...")
y_pred = model.predict(X_test)
y_pred_proba = model.predict_proba(X_test)[:, 1]

print("\n" + "="*60)
print(" MODEL PERFORMANCE EVALUATION")
print("="*60)

accuracy = accuracy_score(y_test, y_pred)
precision = precision_score(y_test, y_pred)
recall = recall_score(y_test, y_pred)
f1 = f1_score(y_test, y_pred)
roc_auc = roc_auc_score(y_test, y_pred_proba)

print(f"\n{'Metric':<15} {'Score':<10} {'Interpretation':<30}")
print("-"*55)
print(f"{'Accuracy':<15} {accuracy:.4f}     {'Overall correct predictions':<30}")
print(f"{'Precision':<15} {precision:.4f}     {'Fraud predictions that were correct':<30}")
print(f"{'Recall':<15} {recall:.4f}     {'Actual frauds detected':<30}")
print(f"{'F1-Score':<15} {f1:.4f}     {'Harmonic mean of P & R':<30}")
print(f"{'ROC-AUC':<15} {roc_auc:.4f}     {'Model discrimination ability':<30}")


cm = confusion_matrix(y_test, y_pred)
print(f"\n Confusion Matrix:")
print(f"                 Predicted")
print(f"               Legit  Fraud")
print(f"Actual Legit   {cm[0,0]:>6} {cm[0,1]:>6}")
print(f"       Fraud    {cm[1,0]:>6} {cm[1,1]:>6}")

print(f"\n Key Insights:")
print(f"    True Positives (Frauds caught): {cm[1,1]:,}")
print(f"    False Negatives (Missed frauds): {cm[1,0]:,}")
print(f"    False Positives (Wrong alerts): {cm[0,1]:,}")
print(f"    Detection Rate: {recall*100:.2f}%")

print(f"\n Detailed Classification Report:")
print(classification_report(y_test, y_pred, target_names=['Legitimate', 'Fraudulent']))

print("\n Generating visualizations...")

fig, axes = plt.subplots(2, 2, figsize=(14, 10))

sns.heatmap(cm, annot=True, fmt='d', ax=axes[0,0], cmap='Blues')
axes[0,0].set_title('Confusion Matrix')
axes[0,0].set_xlabel('Predicted')
axes[0,0].set_ylabel('Actual')
axes[0,0].set_xticklabels(['Legitimate', 'Fraudulent'])
axes[0,0].set_yticklabels(['Legitimate', 'Fraudulent'])

fpr, tpr, _ = roc_curve(y_test, y_pred_proba)
axes[0,1].plot(fpr, tpr, linewidth=2, label=f'Random Forest (AUC = {roc_auc:.3f})')
axes[0,1].plot([0, 1], [0, 1], 'k--', label='Random Classifier')
axes[0,1].set_xlabel('False Positive Rate')
axes[0,1].set_ylabel('True Positive Rate')
axes[0,1].set_title('ROC Curve')
axes[0,1].legend()
axes[0,1].grid(True)

feature_importance = pd.DataFrame({
    'feature': X.columns[:15],
    'importance': model.feature_importances_[:15]
}).sort_values('importance', ascending=True)

axes[1,0].barh(feature_importance['feature'], feature_importance['importance'])
axes[1,0].set_title('Top 15 Feature Importances')
axes[1,0].set_xlabel('Importance Score')

legit_probs = y_pred_proba[y_test == 0][:1000]
fraud_probs = y_pred_proba[y_test == 1]
axes[1,1].hist(legit_probs, bins=30, alpha=0.7, label='Legitimate', density=True)
axes[1,1].hist(fraud_probs, bins=30, alpha=0.7, label='Fraudulent', density=True)
axes[1,1].set_xlabel('Predicted Probability of Fraud')
axes[1,1].set_ylabel('Density')
axes[1,1].set_title('Probability Distribution by Class')
axes[1,1].legend()

plt.tight_layout()
plt.show()

print("\n Finding optimal threshold for fraud detection...")

from sklearn.metrics import precision_recall_curve

precisions, recalls, thresholds = precision_recall_curve(y_test, y_pred_proba)
f1_scores = 2 * (precisions[:-1] * recalls[:-1]) / (precisions[:-1] + recalls[:-1] + 1e-10)
best_threshold = thresholds[np.argmax(f1_scores)]

print(f"Optimal threshold: {best_threshold:.3f}")
print(f"At this threshold, F1-Score would be: {max(f1_scores):.4f}")

print("\n" + "="*60)
print(" CREDIT CARD FRAUD DETECTION SYSTEM SUMMARY")
print("="*60)

print(f"""
 Model Performance:
   • Detection Rate (Recall): {recall*100:.2f}%
   • Precision: {precision*100:.2f}%
   • F1-Score: {f1:.4f}
   • ROC-AUC: {roc_auc:.4f}

 Business Impact:
   • Out of {y_test.sum():.0f} actual frauds, model detected {cm[1,1]:.0f}
   • Missed frauds (False Negatives): {cm[1,0]:.0f}
   • False alerts (False Positives): {cm[0,1]:,} out of {(y_test==0).sum():,} legit transactions

 Recommendations:
   1. Use threshold {best_threshold:.3f} for optimal F1-Score
   2. For higher security, reduce threshold to catch more frauds
   3. For better UX, increase threshold to reduce false alarms
   4. Monitor model performance weekly for concept drift
